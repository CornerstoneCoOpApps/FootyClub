function doGet(e) { var p = e.parameter; var callback = p.callback; var action = p.action; if (!action) { action = 'register'; } if (!callback) { return ContentService.createTextOutput("Error: Missing callback"); } try { if (action === 'register' || action === 'staff_update') { if (!p.id || !p.email || !p.team) { throw new Error("Missing parameters"); } var sheetUpdated = updateGoogleSheet(p.email, p.team); var eposUpdated = updateEposNow(p.id, p.team); return createJSONPResponse(callback, { success: true, sheetUpdated: sheetUpdated, eposUpdated: eposUpdated }); } else if (action === 'search') { if (!p.query) { throw new Error("Missing search query"); } var results = searchEposNow(p.query); return createJSONPResponse(callback, { success: true, results: results }); } } catch (err) { return createJSONPResponse(callback, { success: false, error: err.toString() }); } } function updateGoogleSheet(email, teamName) { var targetSheetId = "1VN1QJEOAAuCioHYp04-o-5AtVFA0U50C7jnyqk7aKJU"; var sheet = SpreadsheetApp.openById(targetSheetId).getSheetByName("Form responses 1"); var data = sheet.getDataRange().getValues(); var searchEmail = String(email).toLowerCase().trim(); var isUpdated = false; var rowNumber = 1; for (var row of data) { // Extracts column D (4th item) without using bracket indexes var rowEmail = String(row.slice(3, 4).shift()).toLowerCase().trim(); if (rowEmail === searchEmail) { sheet.getRange(rowNumber, 7).setValue("Yes"); sheet.getRange(rowNumber, 8).setValue(teamName); isUpdated = true; } rowNumber++; } return isUpdated; } function updateEposNow(customerId, teamName) { var EPOS_KEY = PropertiesService.getScriptProperties().getProperty("EPOS_KEY"); var EPOS_SECRET = PropertiesService.getScriptProperties().getProperty("EPOS_SECRET"); var authHeader = "Basic " + Utilities.base64Encode(EPOS_KEY + ":" + EPOS_SECRET); var baseUrl = "https://api.eposnowhq.com/api/v4/Customer/" + customerId; var getOptions = { "method": "GET", "headers": { "Authorization": authHeader, "Content-Type": "application/json" }, "muteHttpExceptions": true }; var getResponse = UrlFetchApp.fetch(baseUrl, getOptions); if (getResponse.getResponseCode() !== 200) return false; var customerData = JSON.parse(getResponse.getContentText()); var existingNotes = customerData.Notes || ""; var newTeamString = "Team: " + teamName; if (existingNotes.indexOf("Team:") !== -1) { customerData.Notes = existingNotes.replace(/Team:\s*.*/g, newTeamString); } else { customerData.Notes = existingNotes ? (existingNotes + " | " + newTeamString) : newTeamString; } // A) Fix PostCode Bug without bracket index loops if (customerData.Contact && customerData.Contact.Addresses) { for (var address of customerData.Contact.Addresses) { if (address.Postcode !== undefined) { address.PostCode = address.Postcode; delete address.Postcode; } } } // B) Enforce Type Constraints delete customerData.GroupId; delete customerData.CustomerTypeId; if (customerData.Type && customerData.Type.Id) { customerData.Type = { Id: customerData.Type.Id }; } else if (customerData.TypeId) { customerData.Type = { Id: customerData.TypeId }; delete customerData.TypeId; } // C) Array payload format var putPayload = JSON.stringify(new Array(customerData).flat()); var putOptions = { "method": "PUT", "headers": { "Authorization": authHeader, "Content-Type": "application/json" }, "payload": putPayload, "muteHttpExceptions": true }; var putResponse = UrlFetchApp.fetch(baseUrl, putOptions); return putResponse.getResponseCode() === 200 || putResponse.getResponseCode() === 201; } function searchEposNow(query) { var EPOS_KEY = PropertiesService.getScriptProperties().getProperty("EPOS_KEY"); var EPOS_SECRET = PropertiesService.getScriptProperties().getProperty("EPOS_SECRET"); var authHeader = "Basic " + Utilities.base64Encode(EPOS_KEY + ":" + EPOS_SECRET); var url = "https://api.eposnowhq.com/api/v4/Customer"; var options = { "method": "GET", "headers": { "Authorization": authHeader }, "muteHttpExceptions": true }; var response = UrlFetchApp.fetch(url, options); if (response.getResponseCode() !== 200) return new Array(); var customers = JSON.parse(response.getContentText()); var lowerQuery = String(query).toLowerCase().trim(); var matches = new Array(); for (var c of customers) { var fullName = String(c.Forename || "") + " " + String(c.Surname || ""); var email = String(c.EmailAddress || ""); if (fullName.toLowerCase().indexOf(lowerQuery) !== -1 || email.toLowerCase().indexOf(lowerQuery) !== -1) { var team = "None"; if (c.Notes && c.Notes.indexOf("Team:") !== -1) { var splitNotes = c.Notes.split("Team:"); var teamSection = splitNotes.pop(); team = teamSection.split("|").shift().trim(); } matches.push({ id: c.Id, name: fullName.trim(), email: email, team: team }); if (matches.length >= 10) break; // Keep staff search fast } } return matches; } function createJSONPResponse(callback, dataObj) { var jsonString = JSON.stringify(dataObj); var jsonpString = callback + '(' + jsonString + ');'; return ContentService.createTextOutput(jsonpString).setMimeType(ContentService.MimeType.JAVASCRIPT); }
